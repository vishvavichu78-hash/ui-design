function sparkline(vals, color = 'var(--accent)', w = 90, h = 30) {
  const mx = Math.max(...vals), mn = Math.min(...vals), n = vals.length;
  const pts = vals.map((v, i) => `${(i / (n - 1) * w).toFixed(1)},${(h - 3 - (v - mn) / ((mx - mn) || 1) * (h - 6)).toFixed(1)}`).join(' ');
  return `<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" aria-hidden="true"><polyline points="${pts}" fill="none" stroke="${color}" stroke-width="1.6" stroke-linejoin="round"/></svg>`;
}
function areaChart(pts, budget) {
  const W = 800, H = 260, L = 44, B = 24, T = 10, R = 8, n = pts.length;
  const mx = Math.max(...pts.map(p => p.v), budget) * 1.12;
  const x = i => L + i / (n - 1) * (W - L - R), y = v => T + (1 - v / mx) * (H - T - B);
  const line = pts.map((p, i) => `${i ? 'L' : 'M'}${x(i).toFixed(1)},${y(p.v).toFixed(1)}`).join('');
  let g = '';
  for (let k = 0; k <= 4; k++) { const v = mx * k / 4; g += `<line x1="${L}" x2="${W - R}" y1="${y(v)}" y2="${y(v)}" stroke="var(--grid)"/><text x="${L - 6}" y="${y(v) + 3}" text-anchor="end">${v >= 1000 ? (v / 1000).toFixed(1) + 'k' : Math.round(v)}</text>`; }
  const step = Math.max(1, Math.floor(n / 6));
  let xl = '';
  pts.forEach((p, i) => { if (i % step === 0) xl += `<text x="${x(i)}" y="${H - 6}" text-anchor="middle">${p.label}</text>`; });
  const cw = (W - L - R) / (n - 1);
  const hits = pts.map((p, i) => {
    const prev = i ? pts[i - 1].v : p.v;
    return `<rect class="hit" x="${x(i) - cw / 2}" y="${T}" width="${cw}" height="${H - T - B}" data-tip="${esc(p.label)}\n${usd(p.v)}  (${pct((p.v / prev - 1) * 100)})"/>`;
  }).join('');
  return `<svg class="chart" viewBox="0 0 ${W} ${H}" role="img" aria-label="Spend over time">
    <defs><linearGradient id="ag" x1="0" x2="0" y1="0" y2="1"><stop offset="0" stop-color="var(--accent)" stop-opacity=".28"/><stop offset="1" stop-color="var(--accent)" stop-opacity="0"/></linearGradient></defs>
    ${g}${xl}<path d="${line}L${x(n - 1)},${H - B}L${x(0)},${H - B}Z" fill="url(#ag)"/>

    <text x="${W - R}" y="${y(budget) - 5}" text-anchor="end" style="fill:var(--bad)">budget</text>${hits}</svg>`;
}
function donut(items) {
  const total = sum(items.map(i => i.v)), r = 62, c = 2 * Math.PI * r;
  let off = 0;
  const arcs = items.map(it => {
    const len = it.v / total * c;
    const s = `<circle r="${r}" cx="90" cy="90" fill="none" stroke="${it.c}" stroke-width="24" stroke-dasharray="${len - 1.5} ${c - len + 1.5}" stroke-dashoffset="${-off}" transform="rotate(-90 90 90)" data-tip="${it.n}\n${usd(it.v)}  ${(it.v / total * 100).toFixed(1)}%"/>`;
    off += len; return s;
  }).join('');
  return `<svg viewBox="0 0 180 180" class="chart" style="max-width:190px;margin:0 auto" role="img" aria-label="Spend by service">${arcs}
    <text x="90" y="86" text-anchor="middle" style="font-size:11px">total</text><text x="90" y="104" text-anchor="middle" style="font-size:15px;fill:var(--t1)">${usd(total)}</text></svg>`;
}
function barChart(months, mode) {
  const W = 800, H = 280, L = 48, B = 26, T = 10, R = 8;
  const totals = months.map(m => sum(m.vals));
  const mx = (mode === 'stacked' ? Math.max(...totals) : Math.max(...months.flat().map(m => 0), ...months.map(m => Math.max(...m.vals)))) * 1.12;
  const y = v => T + (1 - v / mx) * (H - T - B);
  const gw = (W - L - R) / months.length;
  let out = '';
  for (let k = 0; k <= 4; k++) { const v = mx * k / 4; out += `<line x1="${L}" x2="${W - R}" y1="${y(v)}" y2="${y(v)}" stroke="var(--grid)"/><text x="${L - 6}" y="${y(v) + 3}" text-anchor="end">${(v / 1000).toFixed(0)}k</text>`; }
  months.forEach((m, i) => {
    const gx = L + i * gw;
    out += `<text x="${gx + gw / 2}" y="${H - 8}" text-anchor="middle">${m.label}</text>`;
    if (mode === 'stacked') {
      let acc = 0; const bw = Math.min(54, gw * 0.5);
      m.vals.forEach((v, s) => { const y1 = y(acc + v), y0 = y(acc); out += `<rect x="${gx + (gw - bw) / 2}" y="${y1}" width="${bw}" height="${Math.max(0, y0 - y1)}" fill="${SCOL[s]}" data-tip="${m.label} · ${SERVICES[s]}\n${usd(v)}"/>`; acc += v; });
    } else {
      const bw = Math.min(14, (gw * 0.8) / 6);
      m.vals.forEach((v, s) => { out += `<rect x="${gx + gw * 0.1 + s * (bw + 2)}" y="${y(v)}" width="${bw}" height="${H - B - y(v)}" rx="2" fill="${SCOL[s]}" data-tip="${m.label} · ${SERVICES[s]}\n${usd(v)}"/>`; });
    }
  });
  return `<svg class="chart" viewBox="0 0 ${W} ${H}" role="img" aria-label="Monthly cost by service">${out}</svg>`;
}

function forecast() {
  const f = envF(), daily = DAILY.map(p => p.v * f);
  const today = new Date(), elapsed = today.getDate(), dim = new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate();
  const recent = daily.slice(-14), n = recent.length;
  const mx = (n - 1) / 2, my = sum(recent) / n;
  const slope = sum(recent.map((v, i) => (i - mx) * (v - my))) / sum(recent.map((_, i) => (i - mx) ** 2));
  const spent = sum(daily.slice(-elapsed));
  let proj = spent;
  for (let k = 1; k <= dim - elapsed; k++) proj += my + slope * (n - 1 - mx + k);
  return { spent, proj, low: proj * 0.93, high: proj * 1.07, elapsed, dim };
}

const LIVE = { cpu: [], mem: [], netIn: [], netOut: [], p50: [], p95: [], timer: null };
(function seedLive() {
  const r = mulberry(9); let a = 48, b = 62, c = 40, d = 22, e = 120;
  for (let i = 0; i < 60; i++) { a = clamp(a + (r() - 0.5) * 9, 12, 98); b = clamp(b + (r() - 0.5) * 4, 30, 92); c = clamp(c + (r() - 0.5) * 10, 5, 95); d = clamp(d + (r() - 0.5) * 8, 4, 90); e = clamp(e + (r() - 0.5) * 24, 60, 480);
    LIVE.cpu.push(a); LIVE.mem.push(b); LIVE.netIn.push(c); LIVE.netOut.push(d); LIVE.p50.push(e * 0.55); LIVE.p95.push(e); }
})();
function liveTick() {
  const step = (arr, w, lo, hi) => { arr.push(clamp(arr[arr.length - 1] + (Math.random() - 0.5) * w, lo, hi)); arr.shift(); };
  step(LIVE.cpu, 10, 12, 99); step(LIVE.mem, 4, 30, 94); step(LIVE.netIn, 10, 5, 95); step(LIVE.netOut, 8, 4, 90);
  const e = clamp(LIVE.p95[59] + (Math.random() - 0.5) * 30, 60, 480);
  LIVE.p95.push(e); LIVE.p95.shift(); LIVE.p50.push(e * 0.55); LIVE.p50.shift();
  paintLive();
}

  ['cpu', 'CPU utilisation (%)', 85, () => [{ d: LIVE.cpu, c: 'var(--accent)' }], 100],

  ['net', 'Network (MB/s)  in / out', 80, () => [{ d: LIVE.netIn, c: 'var(--info)' }, { d: LIVE.netOut, c: '#B78AF7' }], 100],
  ['lat', 'Latency (ms)  p50 / p95', 350, () => [{ d: LIVE.p50, c: 'var(--ok)' }, { d: LIVE.p95, c: 'var(--warn)' }], 0]
];

  });
}

function toCsv() {
  const rows = filteredResources();
  const head = ['name', 'id', 'type', 'region', 'status', 'cpu', 'monthly_cost', 'created'];
  const lines = [head.join(',')].concat(rows.map(r => [r.name, r.id, r.type, r.region, r.status, r.cpu, (r.cost * envF()).toFixed(2), new Date(r.created).toISOString().slice(0, 10)].join(',')));
  const a = document.createElement('a');
  a.href = URL.createObjectURL(new Blob([lines.join('\n')], { type: 'text/csv' })); a.download = 'cloudscope-resources.csv'; a.click();
  toast(`Exported ${rows.length} resources`);
}
function resAction(kind, ids) {
  const run = () => {
    if (kind === 'delete') resources = resources.filter(r => !ids.includes(r.id));
    else resources.forEach(r => { if (ids.includes(r.id)) { r.status = kind === 'stop' ? 'stopped' : 'running'; r.cpu = kind === 'stop' ? 0 : 12 + Math.floor(Math.random() * 40); } });
    ids.forEach(i => S.res.sel.delete(i));
    refreshRes(); toast(`${kind === 'delete' ? 'Deleted' : kind === 'stop' ? 'Stopped' : 'Restarted'} ${ids.length} resource${ids.length > 1 ? 's' : ''}`);
  };
  if (kind === 'delete') modal('Delete resource' + (ids.length > 1 ? 's' : '') + '?', `${ids.length} resource${ids.length > 1 ? 's' : ''} will be removed from this demo. This cannot be undone.`, run, 'Delete');
  else run();
}

  if (e.key === 'Escape') { closeOv(); S.menu = null; renderMenu(); return; }
  if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); openPalette(); return; }
  const typing = /INPUT|SELECT|TEXTAREA/.test(e.target.tagName);
  if (typing || e.ctrlKey || e.metaKey || e.altKey) return;
  if (e.key === '/') { e.preventDefault(); const g = $('#gs'); if (g) g.focus(); return; }
  if (e.key === '?') { openShortcuts(); return; }
  if (e.key === 'g') { S.g = Date.now(); return; }
  if (S.g && Date.now() - S.g < 900) {
    const map = { o: 'overview', r: 'resources', c: 'costs', m: 'monitoring', a: 'alerts', s: 'settings' };
    if (map[e.key]) location.hash = '#/' + map[e.key];
    S.g = 0;

let pal = { q: '', i: 0, items: [] };
function openPalette() {
  pal = { q: '', i: 0, items: [] };
  $('#overlay').innerHTML = `<div class="scrim" style="align-items:start"><div class="palette" role="dialog" aria-modal="true" aria-label="Command palette"><input id="pq" placeholder="Jump to a page or resource…" aria-label="Command palette search" autocomplete="off"><div class="list" id="plist"></div></div></div>`;
  const inp = $('#pq'); inp.focus();
  const paint = () => {
    const q = pal.q.toLowerCase();
    const pages = [...PAGES, { id: 'settings', label: 'Settings' }].map(p => ({ label: p.label, hint: 'Page', go: () => location.hash = '#/' + p.id }));
    const res = resources.map(r => ({ label: r.name, hint: r.region, go: () => { S.res.q = r.name; S.res.page = 1; location.hash = '#/resources'; if (route() === 'resources') renderPage(false); } }));
    pal.items = [...pages, ...res].filter(x => x.label.toLowerCase().includes(q)).slice(0, 8);
    pal.i = clamp(pal.i, 0, Math.max(0, pal.items.length - 1));
    $('#plist').innerHTML = pal.items.map((x, i) => `<button class="p ${i === pal.i ? 'sel' : ''}" data-pi="${i}"><span>${esc(x.label)}</span><span class="meta">${x.hint}</span></button>`).join('') || '<div class="empty">No matches</div>';
  };
  inp.addEventListener('input', () => { pal.q = inp.value; pal.i = 0; paint(); });
  inp.addEventListener('keydown', e => {
    if (e.key === 'ArrowDown') { pal.i = Math.min(pal.items.length - 1, pal.i + 1); paint(); e.preventDefault(); }
    if (e.key === 'ArrowUp') { pal.i = Math.max(0, pal.i - 1); paint(); e.preventDefault(); }
    if (e.key === 'Enter' && pal.items[pal.i]) { const it = pal.items[pal.i]; closeOv(); it.go(); }
  });
  $('#plist').addEventListener('click', e => { const b = e.target.closest('[data-pi]'); if (b) { const it = pal.items[+b.dataset.pi]; closeOv(); it.go(); } });
  paint();
}
